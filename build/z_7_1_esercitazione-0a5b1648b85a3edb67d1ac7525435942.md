### `VediImmagini.py`

```python
# -*- coding: utf-8 -*-
"""
Created on Thu Apr 10 12:55:03 2025

@author: mfilo
"""

import numpy as np
import matplotlib.pyplot as plt
import pydicom
import os

# Visualizzazione immagini di dati PET [15O]H2O (per analisi di flusso)


# Fetta n. 15 per 20 time-steps
nome_file = 'H2O_Slice15_time'
time = np.zeros(20)  # vettore tempo
plt.figure()

for n_imm in range(20):
    nome_file_completo = f"{nome_file}{n_imm}.dcm"
    try:
        ds = pydicom.dcmread(nome_file_completo)
        imm = ds.pixel_array.astype(float)  # Converti in float per le operazioni
        inf = ds

        time[n_imm] = (inf.ActualFrameDuration / 1000)  # durata di acquisizione in secondi
        print(f"imm N. {n_imm}, durata acq: {time[n_imm]:.3f} s.")

        # Rescaling dei dati (per avere valori in Bq/ml)
        imm = (imm * inf.RescaleSlope + inf.RescaleIntercept)
        # N.B. senza il rescaling, i valori vanno da 0 a 32767: non sono
        # valori assoluti!!!

        plt.imshow(imm, cmap='gray')
        plt.axis('off')
        plt.colorbar()
        plt.pause(0.3)

    except FileNotFoundError:
        print(f"Errore: File non trovato - {nome_file_completo}")
        break  # Interrompi il loop se un file non viene trovato

time = np.cumsum(time)  # N.B.: ho così generato l'asse dei tempi per le curve TAC

print("\nVettore dei tempi cumulativi (in s.):")
print(time)

plt.show()
```
